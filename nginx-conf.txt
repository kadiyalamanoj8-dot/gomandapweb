# =============================================
# Gomandap Nginx Configuration
# Place at: /etc/nginx/conf.d/gomandap.conf
# =============================================

# Redirect www → non-www
server {
    listen 80;
    server_name www.gomandap.com;
    return 301 http://gomandap.com$request_uri;
}

# Main client site
server {
    listen 80;
    server_name gomandap.com;

    root /home/opc/gomandapweb/client/dist;
    index index.html;

    # Permissions fix: ensure nginx can read the files
    # Run: chmod -R 755 /home/opc/gomandapweb/client/dist
    # Run: chmod 755 /home/opc /home/opc/gomandapweb

    # Enable Gzip compression
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_proxied any;
    gzip_types text/plain text/css text/xml text/javascript application/x-javascript application/xml application/javascript application/json;
    gzip_disable "MSIE [1-6]\.";

    # SPA routing
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Cache static assets
    location ~* \.(?:ico|css|js|gif|jpe?g|png|webp|woff2?|eot|ttf|svg)$ {
        expires 6M;
        access_log off;
        add_header Cache-Control "public, max-age=15552000, immutable";
    }

    # Custom error pages
    error_page 403 404 /index.html;
}

# API backend (reverse proxy to Node.js on port 5000)
server {
    listen 80;
    server_name api.gomandap.com;
    client_max_body_size 50M;

    location / {
        # Handle CORS preflight
        if ($request_method = 'OPTIONS') {
            add_header 'Access-Control-Allow-Origin' $http_origin always;
            add_header 'Access-Control-Allow-Methods' 'GET, POST, PUT, PATCH, DELETE, OPTIONS' always;
            add_header 'Access-Control-Allow-Headers' 'Authorization, Content-Type, Accept, X-Requested-With' always;
            add_header 'Access-Control-Allow-Credentials' 'true' always;
            add_header 'Access-Control-Max-Age' 1728000;
            add_header 'Content-Type' 'text/plain; charset=utf-8';
            add_header 'Content-Length' 0;
            return 204;
        }

        proxy_pass http://localhost:5000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Timeouts
        proxy_connect_timeout 60s;
        proxy_send_timeout    60s;
        proxy_read_timeout    60s;
    }
}
